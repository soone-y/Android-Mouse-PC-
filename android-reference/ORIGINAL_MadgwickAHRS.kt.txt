package com.example.movingsensorapp.sensor

import kotlin.math.sqrt

/**
 * Implementation of Madgwick's IMU and AHRS algorithms.
 * Original C code by Sebastian Madgwick.
 */
class MadgwickAHRS(private var samplePeriod: Float, private var beta: Float = 0.1f) {
    // Quaternion representing estimated orientation
    var q0: Float = 1.0f
    var q1: Float = 0.0f
    var q2: Float = 0.0f
    var q3: Float = 0.0f

    /**
     * Updates the AHRS state with new IMU data.
     * @param gx Gyroscope X axis (rad/s)
     * @param gy Gyroscope Y axis (rad/s)
     * @param gz Gyroscope Z axis (rad/s)
     * @param ax Accelerometer X axis
     * @param ay Accelerometer Y axis
     * @param az Accelerometer Z axis
     * @param dt Time since last update in seconds (if dynamic)
     */
    fun updateIMU(gx: Float, gy: Float, gz: Float, ax: Float, ay: Float, az: Float, dt: Float = samplePeriod) {
        var recipNorm: Float
        var s0: Float; var s1: Float; var s2: Float; var s3: Float
        var qDot1: Float; var qDot2: Float; var qDot3: Float; var qDot4: Float
        var _2q0: Float; var _2q1: Float; var _2q2: Float; var _2q3: Float
        var _4q0: Float; var _4q1: Float; var _4q2: Float; var _8q1: Float; var _8q2: Float
        var q0q0: Float; var q1q1: Float; var q2q2: Float; var q3q3: Float

        // Rate of change of quaternion from gyroscope
        qDot1 = 0.5f * (-q1 * gx - q2 * gy - q3 * gz)
        qDot2 = 0.5f * (q0 * gx + q2 * gz - q3 * gy)
        qDot3 = 0.5f * (q0 * gy - q1 * gz + q3 * gx)
        qDot4 = 0.5f * (q0 * gz + q1 * gy - q2 * gx)

        // Compute feedback only if accelerometer measurement valid
        if (!((ax == 0.0f) && (ay == 0.0f) && (az == 0.0f))) {

            // Normalise accelerometer measurement
            recipNorm = 1.0f / sqrt(ax * ax + ay * ay + az * az)
            val axN = ax * recipNorm
            val ayN = ay * recipNorm
            val azN = az * recipNorm

            // Auxiliary variables
            _2q0 = 2.0f * q0; _2q1 = 2.0f * q1; _2q2 = 2.0f * q2; _2q3 = 2.0f * q3
            _4q0 = 4.0f * q0; _4q1 = 4.0f * q1; _4q2 = 4.0f * q2
            _8q1 = 8.0f * q1; _8q2 = 8.0f * q2
            q0q0 = q0 * q0; q1q1 = q1 * q1; q2q2 = q2 * q2; q3q3 = q3 * q3

            // Gradient decent algorithm corrective step
            s0 = _4q0 * q2q2 + _2q2 * axN + _4q0 * q1q1 - _2q1 * ayN
            s1 = _4q1 * q3q3 - _2q3 * axN + 4.0f * q0q0 * q1 - _2q0 * ayN - _4q1 + _8q1 * q1q1 + _8q1 * q2q2 + _4q1 * azN
            s2 = 4.0f * q0q0 * q2 + _2q0 * axN + _4q2 * q3q3 - _2q3 * ayN - _4q2 + _8q2 * q1q1 + _8q2 * q2q2 + _4q2 * azN
            s3 = 4.0f * q1q1 * q3 - _2q1 * axN + 4.0f * q2q2 * q3 - _2q2 * ayN

            // Normalise step magnitude
            recipNorm = 1.0f / sqrt(s0 * s0 + s1 * s1 + s2 * s2 + s3 * s3)
            s0 *= recipNorm; s1 *= recipNorm; s2 *= recipNorm; s3 *= recipNorm

            // Apply feedback step
            qDot1 -= beta * s0
            qDot2 -= beta * s1
            qDot3 -= beta * s2
            qDot4 -= beta * s3
        }

        // Integrate rate of change of quaternion
        q0 += qDot1 * dt
        q1 += qDot2 * dt
        q2 += qDot3 * dt
        q3 += qDot4 * dt

        // Normalise quaternion
        recipNorm = 1.0f / sqrt(q0 * q0 + q1 * q1 + q2 * q2 + q3 * q3)
        q0 *= recipNorm
        q1 *= recipNorm
        q2 *= recipNorm
        q3 *= recipNorm
    }

    /**
     * Extracts the gravity vector from the current quaternion.
     * The returned vector is normalized (length 1).
     * Multiply by 9.80665 to get it in m/s^2.
     */
    fun getGravityVector(): FloatArray {
        val gx = 2.0f * (q1 * q3 - q0 * q2)
        val gy = 2.0f * (q0 * q1 + q2 * q3)
        val gz = q0 * q0 - q1 * q1 - q2 * q2 + q3 * q3
        return floatArrayOf(gx, gy, gz)
    }
}
